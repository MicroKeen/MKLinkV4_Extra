cmake_minimum_required(VERSION 3.20)
project(fishing_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(APP "${CMAKE_CURRENT_SOURCE_DIR}/../src/fish")
set(CORE ${APP}/fish_game.c ${APP}/fish_economy.c ${APP}/fish_species.c ${APP}/fish_species_data.c
    ${APP}/fish_map.c ${APP}/fish_map_data.c ${APP}/fish_equipment.c ${APP}/fish_equipment_data.c)
file(GLOB CRYPTO "${APP}/crypto/source/*.c")
foreach(name game fishing fight_balance rarity map species economy variant content_audit)
    add_executable(${name}_tests ${name}_tests.c ${CORE})
    target_include_directories(${name}_tests PRIVATE "${APP}")
    target_compile_options(${name}_tests PRIVATE -UNDEBUG)
    add_test(NAME ${name} COMMAND ${name}_tests)
endforeach()
add_executable(save_tests save_tests.c ${CORE} ${APP}/fish_save.c ${APP}/fish_save_crypto.c ${APP}/fish_save_key.c ${CRYPTO})
target_include_directories(save_tests PRIVATE "${APP}" "${APP}/crypto/include")
target_compile_options(save_tests PRIVATE -UNDEBUG)
add_test(NAME encrypted_save COMMAND save_tests)
add_executable(input_test input_test.c ${CORE} ${APP}/fish_hpm_input.c
    ../src/key_module/key_module.c ../src/key_module/key_task.c)
target_include_directories(input_test PRIVATE input_stubs "${APP}" ../src/key_module)
target_compile_options(input_test PRIVATE -UNDEBUG)
add_test(NAME encoder_to_game COMMAND input_test)
add_executable(sd_backlight_tests sd_backlight_tests.c ../lcd/adapt/spi_sd_adapt.c)
target_include_directories(sd_backlight_tests PRIVATE sd_hal_stubs ../lcd/adapt)
target_compile_options(sd_backlight_tests PRIVATE -UNDEBUG)
add_test(NAME sd_backlight_isolation COMMAND sd_backlight_tests)

add_executable(feedback_tests feedback_tests.c ${CORE} ${APP}/fish_hpm_feedback.c)
target_include_directories(feedback_tests PRIVATE feedback_stubs "${APP}")
target_compile_options(feedback_tests PRIVATE -UNDEBUG -Wall -Wextra)
add_test(NAME buzzer_feedback COMMAND feedback_tests)
